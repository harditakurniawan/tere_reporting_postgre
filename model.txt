DROP TABLE IF EXISTS "mongo"."report_redeem_transaction";
CREATE TABLE "mongo"."report_redeem_transaction" (
    "_id" varchar(24) NOT NULL,
    "transaction_date" timestamptz NOT NULL,
    "transaction_id" varchar(64),
    "keyword" varchar(24),
    "execution_type" varchar(24),
    "product_id" varchar(64),
    "period1" varchar(24),
    "period2" varchar(24),
    "subscriber_id" varchar(64),
    "msisdn" varchar(24),
    "return_value" varchar(24),
    "execution_date" timestamptz NOT NULL,
    "channel_code" varchar(24),
    "transaction_status" varchar(24),
    "trdm_last_act" varchar(24),
    "trdm_act_status" int4,
    "trdm_evd_id" varchar(64),
    "trdm_flag_kirim" varchar(24),
    "trdm_geneva_exec" varchar(24),
    "trdm_keyword" varchar(24),
    "trdm_tgl_kirim" varchar(24),
    "channel_transaction_id" varchar(24),
    "card_type" varchar(24),
    "brand" varchar(24),
    "subscriber_region" varchar(24),
    PRIMARY KEY ("_id")
);

-- Indexes
CREATE INDEX transaction_id_index ON mongo.report_redeem_transaction USING btree (transaction_id);
CREATE INDEX msisdn_index ON mongo.report_redeem_transaction USING btree (msisdn);
CREATE INDEX keyword_index ON mongo.report_redeem_transaction USING btree (keyword);
CREATE INDEX transaction_date_index ON mongo.report_redeem_transaction USING btree (transaction_date);
